======================================================
Answers are in practice_answers.txt. Try first, then check.


SECTION A - PREDICT THE OUTPUT
------------------------------
A1.  x = 15
     if x > 20:
         print("big")
     elif x > 10:
         print("medium")
     else:
         print("small")

A2.  total = 12.0
     items = 2
     if total > 10.0:
         if items > 3:
             print("D")
     else:
         print("E")

A3.  for n in range(3):
         print(n, end=" ")

A4.  for n in range(2, 11, 4):
         print(n, end=" ")

A5.  for n in range(5, 0, -2):
         print(n, end=" ")

A6.  count = 0
     while count < 4:
         count = count + 1
     print(count)

A7.  count = 0
     while count <= 4:
         count = count + 1
     print(count)

A8.  total = 0
     for n in range(1, 5):
         total = total + n
     print(total)

A9.  word = "banana"
     a_count = 0
     for letter in word:
         if letter == "a":
             a_count = a_count + 1
     print(a_count)

A10. print("cat" in "concatenate", "dog" in "hotdog stand", "Hot" in "hotdog")
