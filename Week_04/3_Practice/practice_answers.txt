C4.
    def biggest_name(names, ages):
        """Return the name of the oldest person."""
        best = 0
        for i in range(len(ages)):
            if ages[i] > ages[best]:     # > for biggest (5b used <)
                best = i
        return names[best]

    print(biggest_name(["ana", "bob", "cy"], [20, 41, 33]))   # bob

    print("Only Python:", python_class - excel_class)   # {'ana', 'cy'}
